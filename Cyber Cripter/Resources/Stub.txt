    class Decryption

        public static byte[] XOR(byte[] data, string key)
        {
            byte[] bKey = Encoding.ASCII.GetBytes(key);
            int dataLen = data.Length;
            int keyLen = bKey.Length;
            byte[] output = new byte[dataLen];

            for (int i = 0; i < dataLen; ++i)
            {
                output[i] = (byte)(data[i] ^ bKey[i % keyLen]);
            }

            return output;
        }

        public static byte[] RC4(byte[] input, string key)
        {
            byte[] bKey = Encoding.ASCII.GetBytes(key);
            int dataLen = input.Length;
            byte[] result = new byte[dataLen];
            int x, y, j = 0;
            int[] box = new int[256];

            for (int i = 0; i < 256; i++)
            {
                box[i] = i;
            }

            for (int i = 0; i < 256; i++)
            {
                j = (bKey[i % bKey.Length] + box[i] + j) % 256;
                x = box[i];
                box[i] = box[j];
                box[j] = x;
            }

            for (int i = 0; i < input.Length; i++)
            {
                y = i % 256;
                j = (box[y] + j) % 256;
                x = box[y];
                box[y] = box[j];
                box[j] = x;

                result[i] = ((byte)(input[i] ^ box[(box[y] + box[j]) % 256]));
            }
            return result;
        }
